# ui/theme_builder.gd
static func build(pal: Dictionary) -> Theme:
	var th := Theme.new()
	var regular: Font = _font("res://assets/fonts/Kanit-Regular.ttf")
	th.default_font = regular
	th.default_font_size = 20
	var panel := StyleBoxFlat.new()
	panel.bg_color = Color(pal["ui_bg"])
	panel.set_corner_radius_all(10)
	panel.set_content_margin_all(12)
	th.set_stylebox("panel", "PanelContainer", panel)
	th.set_color("font_color", "Label", Color(pal["ui_text"]))
	# ... buttons: normal / hover / pressed / disabled / focus ...
	return th

static func _font(path: String) -> Font:
	if ResourceLoader.exists(path):
		return load(path) as Font
	push_warning("font missing: " + path)
	return ThemeDB.fallback_font
