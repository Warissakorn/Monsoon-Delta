# ui/audio.gd
func _ensure_bus(bus_name: String) -> void:
	if AudioServer.get_bus_index(bus_name) != -1:
		return
	AudioServer.add_bus()
	var idx: int = AudioServer.bus_count - 1
	AudioServer.set_bus_name(idx, bus_name)
	AudioServer.set_bus_send(idx, "Master")

func set_bus_linear(bus_name: String, v: float) -> void:
	var idx: int = AudioServer.get_bus_index(bus_name)
	AudioServer.set_bus_mute(idx, v < 0.001)
	AudioServer.set_bus_volume_db(idx, linear_to_db(maxf(v, 0.001)))

func _load_stream(file: String, loop: bool) -> AudioStream:
	var path: String = "res://assets/audio/" + file
	if not ResourceLoader.exists(path):      # NOT FileAccess.file_exists: fails in exported builds
		print_verbose("audio missing: " + path)
		return null
	var s: AudioStream = load(path)
	if s is AudioStreamOggVorbis:
		(s as AudioStreamOggVorbis).loop = loop
	return s

# synthesized rain when amb_rain.ogg is missing
func _start_rain_synth() -> void:
	var gen := AudioStreamGenerator.new()
	gen.mix_rate = 22050.0
	gen.buffer_length = 0.3
	_synth = AudioStreamPlayer.new()
	_synth.stream = gen
	_synth.bus = "Ambience"
	_synth.volume_db = -18.0
	add_child(_synth)
	_synth.play()
	_playback = _synth.get_stream_playback() as AudioStreamGeneratorPlayback

func _process(_delta: float) -> void:
	if _playback == null:
		return
	for i in _playback.get_frames_available():
		var x: float = _noise_rng.randf_range(-1.0, 1.0)   # own RNG, never state.rng
		_lp += 0.08 * (x - _lp)
		_playback.push_frame(Vector2(_lp, _lp))
